|------|-----|-----------|------|------|
| text-embedding-3-large | 3072 | ★★★★★ | 높음 | 정밀도 최우선 |
| text-embedding-3-small | 1536 | ★★★★ | 낮음 | 일반 사용 |
| BAAI/bge-m3 | 1024 | ★★★★★ | 무료 | 로컬 배포 |
| ko-sroberta-multitask | 768 | ★★★ | 무료 | 경량 한국어 |

### 실험 코드

```python
from langchain_openai import OpenAIEmbeddings
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_community.vectorstores import FAISS
from langchain_core.documents import Document

def compare_embedding_models(docs, queries):
    models = {
        "openai-small": OpenAIEmbeddings(model="text-embedding-3-small"),
        "bge-m3": HuggingFaceEmbeddings(model_name="BAAI/bge-m3"),
    }

    results = {}
    for model_name, emb in models.items():
        vectorstore = FAISS.from_documents(docs, emb)
        retriever = vectorstore.as_retriever(search_kwargs={"k": 3})

        hit_rates = []
        for query, expected in queries:
            retrieved = retriever.invoke(query)
            retrieved_ids = {d.metadata.get("id") for d in retrieved}
            hit = expected in retrieved_ids
            hit_rates.append(hit)

        results[model_name] = {
            "hit_rate": sum(hit_rates) / len(hit_rates)
        }

    return results
```

## 4. 청크 전략 실험 비교

```python
def experiment_chunk_strategies(text, query, embedding_model):
    strategies = [
        ("고정 300자", RecursiveCharacterTextSplitter(chunk_size=300, chunk_overlap=50)),
        ("고정 500자", RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=100)),
        ("고정 1000자", RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)),
        ("의미 기반", SemanticChunker(embedding_model)),
    ]

    for name, splitter in strategies:
        chunks = splitter.split_text(text)
        vectorstore = FAISS.from_texts(chunks, embedding_model)
        results = vectorstore.similarity_search(query, k=3)
        print(f"\n[{name}] 청크 수: {len(chunks)}, 검색 결과 Top1:")
        print(f"  {results[0].page_content[:100]}")
```
