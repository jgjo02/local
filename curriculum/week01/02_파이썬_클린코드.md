## 1️⃣ 클린코드란 무엇인가?

### 1.1 클린코드의 정의

"클린코드는 **읽기 쉬운** 코드입니다." — Robert C. Martin

AI 서비스 개발에서 클린코드가 중요한 이유:
- **팀 협업**: 다른 개발자가 프롬프트 체인을 빠르게 이해해야 합니다
- **유지보수**: LLM 응답 처리 로직은 자주 변경됩니다
- **디버깅**: AI 응답의 불확실성으로 인해 코드 자체의 버그를 명확히 구분해야 합니다

## 4️⃣ 데이터클래스 (dataclass)

### 4.1 기본 데이터클래스

데이터클래스는 AI 서비스에서 설정(Config), 요청(Request), 응답(Response) 모델에 매우 유용합니다.

```python
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class LLMConfig:
    """LLM 설정을 관리하는 데이터클래스"""
    model: str = "gpt-4o-mini"
    temperature: float = 0.7
    max_tokens: int = 2048
    top_p: float = 1.0
    stream: bool = False

    def __post_init__(self):
        """초기화 후 유효성 검사"""
        if not 0 <= self.temperature <= 2:
            raise ValueError(f"temperature는 0-2 사이여야 합니다: {self.temperature}")
        if not 0 < self.top_p <= 1:
            raise ValueError(f"top_p는 0-1 사이여야 합니다: {self.top_p}")


@dataclass
class ChatRequest:
    """채팅 요청 모델"""
    query: str
    session_id: str
    config: LLMConfig = field(default_factory=LLMConfig)
    metadata: dict = field(default_factory=dict)


@dataclass
class ChatResponse:
    """채팅 응답 모델"""
    answer: str
    sources: list[str] = field(default_factory=list)
    tokens_used: int = 0
    latency_ms: float = 0.0
    model: str = ""

# 사용 예시
config = LLMConfig(temperature=0.3, stream=True)
request = ChatRequest(query="안녕하세요", session_id="user-123", config=config)
print(request)
# ChatRequest(query='안녕하세요', session_id='user-123', config=LLMConfig(...), metadata={})
```

```python
from pydantic import BaseModel, Field, validator

class RAGConfig(BaseModel):
    """RAG 파이프라인 설정"""
    chunk_size: int = Field(default=1000, ge=100, le=8000)
    chunk_overlap: int = Field(default=200, ge=0)
    top_k: int = Field(default=4, ge=1, le=20)
    embedding_model: str = "text-embedding-3-small"
    vector_store_type: Literal["faiss", "chroma"] = "faiss"

    @validator("chunk_overlap")
    def overlap_must_be_less_than_chunk(cls, v, values):
        if "chunk_size" in values and v >= values["chunk_size"]:
            raise ValueError("chunk_overlap은 chunk_size보다 작아야 합니다")
        return v

# JSON 직렬화/역직렬화
config = RAGConfig(chunk_size=500, chunk_overlap=100)
```

```python
import logging
from openai import OpenAIError, RateLimitError, APITimeoutError

logger = logging.getLogger(__name__)


# 나쁜 예: 너무 넓은 예외 처리
def bad_chat(query: str) -> str:
    try:
        # ... API 호출
        pass
    except Exception:  # 모든 예외를 잡으면 버그를 숨깁니다
        return ""


# 좋은 예: 구체적인 예외 처리
def good_chat(query: str) -> str | None:
    try:
        # ... API 호출
        pass
    except RateLimitError as e:
        logger.warning(f"Rate limit 초과: {e}. 잠시 후 재시도하세요.")
```

### 5.2 커스텀 예외 클래스

```python
class AIServiceError(Exception):
    """AI 서비스 기본 예외 클래스"""
    pass


class LLMGenerationError(AIServiceError):
    """LLM 생성 실패 예외"""
    def __init__(self, message: str, model: str, retry_count: int = 0):
        super().__init__(message)
        self.model = model
        self.retry_count = retry_count


class DocumentProcessingError(AIServiceError):
    """문서 처리 실패 예외"""
    def __init__(self, message: str, file_path: str):
        super().__init__(message)
        self.file_path = file_path


class VectorStoreError(AIServiceError):
    """벡터저장소 오류"""
    pass


# 사용 예시
def load_document(file_path: str) -> str:
    if not os.path.exists(file_path):
        raise DocumentProcessingError(
            f"파일을 찾을 수 없습니다",
            file_path=file_path
        )
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return f.read()
    except UnicodeDecodeError as e:
        raise DocumentProcessingError(
            f"파일 인코딩 오류: {e}",
            file_path=file_path
        ) from e
```

### 5.3 컨텍스트 매니저를 활용한 예외 처리

```python
from contextlib import contextmanager
import time


@contextmanager
def timer(operation_name: str):
    """코드 블록의 실행 시간을 측정하는 컨텍스트 매니저"""
    start = time.perf_counter()
    try:
        yield
    finally:
        elapsed = (time.perf_counter() - start) * 1000
        logger.info(f"{operation_name} 완료: {elapsed:.2f}ms")


# 사용 예시
with timer("LLM 응답 생성"):
    response = llm.invoke("안녕하세요")
```

### 6.3 제너레이터와 이터레이터

```python
from typing import Generator, Iterator


def chunk_text(
    text: str,
    chunk_size: int,
    overlap: int = 0,
) -> Generator[str, None, None]:
    """텍스트를 청크로 분할하는 제너레이터"""
    start = 0
    while start < len(text):
        end = start + chunk_size
        yield text[start:end]
        start += chunk_size - overlap


# 사용 예시 - 메모리 효율적으로 대용량 문서 처리
large_text = "..." * 100000  # 매우 큰 텍스트

for i, chunk in enumerate(chunk_text(large_text, chunk_size=1000, overlap=100)):
    # 각 청크를 즉시 처리 (전체를 메모리에 로드하지 않음)
    process_chunk(chunk)
    if i % 100 == 0:
        logger.info(f"{i}번째 청크 처리 중...")
```

```python
class Environment(str, Enum):
    DEVELOPMENT = "development"
    STAGING = "staging"
    PRODUCTION = "production"


class DevelopmentSettings(Settings):

    log_level: str = "DEBUG"

    debug: bool = False
```

### 10.1 표준 로깅 설정

```python
import logging
import sys
from pathlib import Path


def setup_logging(
    level: str = "INFO",
    log_file: str | None = None,
) -> logging.Logger:
    """애플리케이션 로깅 설정"""

    # 로그 포맷 정의
    formatter = logging.Formatter(
        fmt="%(asctime)s | %(levelname)-8s | %(name)s:%(lineno)d | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )

    # 루트 로거 설정
    root_logger = logging.getLogger()
    root_logger.setLevel(getattr(logging, level.upper()))

    # 콘솔 핸들러
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(formatter)
    root_logger.addHandler(console_handler)

    # 파일 핸들러 (선택)
    if log_file:
        Path(log_file).parent.mkdir(parents=True, exist_ok=True)
        file_handler = logging.FileHandler(log_file, encoding="utf-8")
        file_handler.setFormatter(formatter)
        root_logger.addHandler(file_handler)

    # 외부 라이브러리 로그 레벨 조정
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("openai").setLevel(logging.WARNING)

    return root_logger


# 모듈별 로거 생성
logger = logging.getLogger(__name__)

# 사용 예시
```

## 💡 핵심 포인트

1. **네이밍이 90%**: 좋은 변수명, 함수명을 사용하면 주석이 필요 없습니다
2. **타입 힌팅은 문서화**: 타입 힌팅을 작성하는 것은 미래의 나를 위한 문서 작성입니다
3. **데이터클래스로 명확성**: 딕셔너리 대신 데이터클래스를 사용하면 IDE 자동완성이 됩니다
4. **환경변수는 필수**: API 키는 절대 코드에 하드코딩하지 않습니다
5. **로깅이 디버깅**: `print()`는 개발 중에만, 실서비스는 반드시 `logging`을 사용합니다

## ❓ 차시별 질문

1. **Q**: `dataclass`와 `Pydantic BaseModel`의 차이는 무엇인가요?
   - **A**: `dataclass`는 표준 라이브러리로 단순한 데이터 컨테이너에 적합합니다. Pydantic은 유효성 검사, 직렬화, JSON 스키마 생성 등 고급 기능을 제공합니다. LangChain은 Pydantic을 기반으로 합니다.

2. **Q**: 타입 힌팅을 작성하면 실행 속도가 느려지나요?
   - **A**: 아니요. Python의 타입 힌팅은 런타임에 무시됩니다. 정적 분석 도구(mypy)가 코드 실행 전에 검사합니다.

3. **Q**: 로깅과 print의 실질적인 차이가 무엇인가요?
   - **A**: 로깅은 레벨 필터링, 파일 저장, 타임스탬프, 멀티스레드 안전성을 지원합니다. print는 개발용 임시 출력에만 사용하세요.

4. **Q**: 가상환경을 반드시 사용해야 하나요?
   - **A**: 네. 프로젝트마다 다른 버전의 패키지가 필요할 수 있고, 전역 Python 환경을 오염시키지 않기 위해 필수입니다.

5. **Q**: 환경변수가 없을 때 기본값을 어떻게 설정하나요?
   - **A**: Pydantic Settings에서 `field(default="기본값")`을 사용하거나, `os.getenv("KEY", "기본값")`을 사용합니다.
