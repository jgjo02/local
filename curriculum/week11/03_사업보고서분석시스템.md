## 1. DART API 연동

### 1.1 DART API 기본

```python
import requests
from typing import Optional
import json

DART_BASE_URL = "https://opendart.fss.or.kr/api"

class DARTClient:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = DART_BASE_URL

    def get_company_code(self, company_name: str) -> Optional[str]:
        """회사명으로 고유번호 조회"""
        url = f"{self.base_url}/company.json"
        params = {
            "crtfc_key": self.api_key,
            "corp_name": company_name,
        }
        response = requests.get(url, params=params)
        data = response.json()

        if data.get("status") == "000" and data.get("corp_code"):
            return data["corp_code"]
        return None

    def get_financial_statements(
        self,
        corp_code: str,
        year: str,
        report_code: str = "11011",  # 사업보고서
    ) -> dict:
        """재무제표 조회"""
        url = f"{self.base_url}/fnlttSinglAcntAll.json"
        params = {
            "crtfc_key": self.api_key,
            "corp_code": corp_code,
            "bsns_year": year,
            "reprt_code": report_code,
            "fs_div": "CFS",  # 연결 재무제표
        }
        response = requests.get(url, params=params)
        return response.json()

    def get_annual_report(self, corp_code: str, year: str) -> dict:
        """사업보고서 원문 정보"""
        url = f"{self.base_url}/list.json"
        params = {
            "crtfc_key": self.api_key,
            "corp_code": corp_code,
            "bgn_de": f"{year}0101",
            "end_de": f"{year}1231",
            "pblntf_ty": "A",  # 사업보고서
        }
        response = requests.get(url, params=params)
        return response.json()
```

### 1.2 재무 데이터 추출

```python
def extract_key_financials(financial_data: dict) -> dict:
    """핵심 재무 지표 추출"""
    result = {}
    items = financial_data.get("list", [])

    key_accounts = {
        "매출액": "revenue",
        "영업이익": "operating_income",
        "당기순이익": "net_income",
        "자산총계": "total_assets",
        "부채총계": "total_liabilities",
        "자본총계": "total_equity",
    }

    for item in items:
        account_name = item.get("account_nm", "")
        for korean, english in key_accounts.items():
            if korean in account_name:
                try:
                    value = int(item.get("thstrm_amount", "0").replace(",", ""))
                    result[english] = value
                except:
                    pass

    # 파생 지표 계산
    if "operating_income" in result and "revenue" in result and result["revenue"] > 0:
        result["operating_margin"] = result["operating_income"] / result["revenue"] * 100

    return result
```

## 2. 재무 분석 프롬프트

```python
FINANCIAL_ANALYSIS_PROMPT = """당신은 재무 분석 전문가입니다.

## 재무 데이터
{financial_data}

## 분석 요청
{analysis_type}

분석 형식:
1. 핵심 수치 요약
2. 전년 대비 변화 (있는 경우)
3. 업계 평균 대비 (가능한 경우)
4. 주요 리스크 요인
5. 종합 의견"""

def analyze_financials(
    company: str,
    financials: dict,
    analysis_type: str = "종합 재무 분석",
    llm=None,
) -> str:
    from langchain_openai import ChatOpenAI
    from langchain_core.messages import HumanMessage

    if llm is None:
        llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.3)

    financial_str = json.dumps(financials, ensure_ascii=False, indent=2)
    prompt = FINANCIAL_ANALYSIS_PROMPT.format(
        financial_data=f"{company} 재무 데이터:\n{financial_str}",
        analysis_type=analysis_type,
    )

    response = llm.invoke([HumanMessage(content=prompt)])
    return response.content
```

## 3. 기업 비교 분석

```python
def compare_companies(
    companies: list,
    metrics: list = None,
    llm=None,
) -> str:
    """여러 기업 재무 비교 분석"""
    if llm is None:
        from langchain_openai import ChatOpenAI
        llm = ChatOpenAI(model="gpt-4o-mini")

    if metrics is None:
        metrics = ["revenue", "operating_income", "operating_margin", "total_assets"]

    # 비교 데이터 테이블 생성
    comparison = {}
    for company_data in companies:
        name = company_data["name"]
        financials = company_data["financials"]
        comparison[name] = {m: financials.get(m, "N/A") for m in metrics}

    comparison_str = json.dumps(comparison, ensure_ascii=False, indent=2)

    prompt = f"""다음 기업들의 재무 지표를 비교 분석해주세요.

{comparison_str}

분석 요청:
1. 매출 규모 비교
2. 수익성 비교 (영업이익률)
3. 재무 건전성 비교
4. 경쟁력 분석
5. 투자 매력도 종합 의견"""

    from langchain_core.messages import HumanMessage
    response = llm.invoke([HumanMessage(content=prompt)])
    return response.content
```

## 4. 사업보고서 RAG 통합

```python
class BusinessReportAnalyzer:
    """사업보고서 종합 분석 시스템"""

    def __init__(self, vector_store, dart_client=None):
        self.vector_store = vector_store
        self.dart_client = dart_client
        from langchain_openai import ChatOpenAI
        self.llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.3)

    def analyze_with_rag(self, company: str, question: str) -> dict:
        """RAG 기반 사업보고서 분석"""
        # 문서 검색
        query = f"{company} {question}"
        docs = self.vector_store.similarity_search(query, k=5)
        context = "\n\n".join([doc.page_content for doc in docs[:3]])

        prompt = f"""다음 사업보고서 내용을 바탕으로 질문에 답하세요.

회사: {company}

관련 내용:
{context}

질문: {question}"""

        from langchain_core.messages import HumanMessage
        response = self.llm.invoke([HumanMessage(content=prompt)])

        return {
            "company": company,
            "question": question,
            "answer": response.content,
            "sources": [doc.metadata for doc in docs[:3]],
        }

    def annual_comparison(self, company: str, years: list) -> str:
        """연도별 실적 비교"""
        query = f"{company} 연도별 실적 매출 영업이익"
        docs = self.vector_store.similarity_search(query, k=8)
        context = "\n\n".join([doc.page_content for doc in docs])

        prompt = f"""{company}의 연도별 실적을 비교 분석해주세요.

관련 자료:
{context}

분석 연도: {', '.join(map(str, years))}

비교 항목:
1. 매출액 추이
2. 영업이익 및 영업이익률 변화
3. 주요 성장/하락 요인
4. 향후 전망"""

        from langchain_core.messages import HumanMessage
        response = self.llm.invoke([HumanMessage(content=prompt)])
        return response.content
```
