### 3.2 에러 처리 패턴

```python
from tenacity import retry, stop_after_attempt, wait_exponential
import openai

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=4, max=10),
    retry=lambda e: isinstance(e, (openai.RateLimitError, openai.APIError)),
)
def robust_llm_call(prompt: str, llm) -> str:
    try:
        return llm.invoke(prompt).content
    except openai.RateLimitError:
        raise  # 재시도 대상
    except openai.BadRequestError as e:
        # 입력 오류 - 재시도 불필요
        return f"입력 오류: {str(e)}"
    except Exception as e:
        return f"예상치 못한 오류: {str(e)}"
```

### 3.3 검색 품질 보증

```python
def validate_retrieval(docs: list, query: str, min_docs: int = 1) -> bool:
    """검색 결과 최소 품질 검증"""
    if len(docs) < min_docs:
        return False

    # 기본적인 관련성 체크 (키워드 포함 여부)
    query_words = set(query.lower().split())
    relevant_count = 0

    for doc in docs:
        doc_words = set(doc.page_content.lower().split())
        if len(query_words & doc_words) > 0:
            relevant_count += 1

    return relevant_count >= min_docs
```

import asyncio
from langchain_openai import ChatOpenAI

async def async_generate(queries: list) -> list:
    llm = ChatOpenAI(model="gpt-4o-mini")
    tasks = [llm.ainvoke(q) for q in queries]
    return await asyncio.gather(*tasks)
