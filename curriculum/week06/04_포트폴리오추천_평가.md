## 2. 수익률 분석 시각화

### 2.1 Plotly 차트 생성

```python
import plotly.graph_objects as go
import plotly.express as px
import pandas as pd

def create_portfolio_chart(portfolio: dict) -> go.Figure:
    """포트폴리오 구성 파이 차트"""
    labels = [item["name"] for item in portfolio]
    values = [item["weight"] for item in portfolio]

    fig = go.Figure(data=[go.Pie(
        labels=labels,
        values=values,
        hole=0.4,
        textinfo="label+percent",
    )])

    fig.update_layout(
        title="포트폴리오 구성",
        showlegend=True,
    )
    return fig

def create_return_comparison(etfs: list) -> go.Figure:
    """ETF 수익률 비교 막대 차트"""
    periods = ["1개월", "3개월", "6개월", "1년", "3년"]
    keys = ["1m_return", "3m_return", "6m_return", "1y_return", "3y_return"]

    fig = go.Figure()
    for etf in etfs[:5]:
        returns = [etf.get(k, 0) for k in keys]
        fig.add_trace(go.Bar(
            name=etf["name"],
            x=periods,
            y=returns,
            text=[f"{r:+.1f}%" for r in returns],
            textposition="auto",
        ))

    fig.update_layout(
        title="ETF 기간별 수익률 비교",
        barmode="group",
        yaxis_title="수익률 (%)",
    )
    return fig
```

## 3. ETF 추천 품질 평가

### 3.1 관련성 평가 (LLM-as-Judge)

```python
RELEVANCE_EVAL_PROMPT = """아래 사용자 질문과 검색된 ETF의 관련성을 평가해주세요.

질문: {question}

ETF 정보:
{etf_info}

평가 기준:
- 5점: 질문의 요구사항을 완벽히 충족
- 4점: 대부분 관련 있음
- 3점: 부분적으로 관련 있음
- 2점: 약간 관련 있음
- 1점: 관련 없음

점수: [숫자만]
이유: [한 문장]"""

def evaluate_relevance(question: str, etf: dict, llm) -> dict:
    from langchain_core.messages import HumanMessage

    prompt = RELEVANCE_EVAL_PROMPT.format(
        question=question,
        etf_info=f"{etf['name']} - {etf['description']}"
    )

    response = llm.invoke([HumanMessage(content=prompt)])

    score = 3  # 기본값
    for line in response.content.split("\n"):
        if line.startswith("점수:"):
            try:
                score = int(line.split(":")[1].strip())
            except:
                pass

    return {"etf": etf["name"], "score": score, "response": response.content}
```

### 3.2 답변 품질 평가

```python
ANSWER_QUALITY_CRITERIA = {
    "accuracy": "제공된 ETF 정보와 답변 내용이 일치하는가?",
    "completeness": "사용자 질문의 모든 요소를 다루었는가?",
    "clarity": "전문 용어를 이해하기 쉽게 설명했는가?",
    "appropriateness": "위험 고지 등 금융 상담 의무를 준수했는가?",
}

def evaluate_answer(question: str, answer: str, llm) -> dict:
    criteria_text = "\n".join(
        f"- {k}: {v}" for k, v in ANSWER_QUALITY_CRITERIA.items()
    )

    eval_prompt = f"""다음 ETF 상담 응답을 평가해주세요.

질문: {question}
답변: {answer}

평가 기준:
{criteria_text}

각 기준별 1-5점 점수와 이유를 JSON으로 반환하세요.
{{"accuracy": {{"score": 4, "reason": "..."}}, ...}}"""

    response = llm.invoke([HumanMessage(content=eval_prompt)])
    return {"raw_evaluation": response.content}
```

### 3.3 배치 평가 실행

```python
test_cases = [
    {
        "question": "노후 대비로 안전한 ETF 추천",
        "expected_risk": "저위험",
        "expected_categories": ["채권", "국내주식"],
    },
    {
        "question": "AI 반도체 테마 고성장 투자",
        "expected_risk": "고위험",
        "expected_categories": ["섹터"],
    },
]

def run_evaluation(test_cases: list, recommender, llm) -> dict:
    results = []
    for case in test_cases:
        result = recommender.recommend(case["question"])
        relevance_scores = [
            evaluate_relevance(case["question"], doc.metadata, llm)
            for doc in result["source_docs"][:3]
        ]
        quality = evaluate_answer(case["question"], result["answer"], llm)

        results.append({
            "question": case["question"],
            "avg_relevance": sum(r["score"] for r in relevance_scores) / len(relevance_scores),
            "quality_eval": quality,
        })

    return {
        "results": results,
        "overall_relevance": sum(r["avg_relevance"] for r in results) / len(results),
    }
```

## 4. 성능 최적화

### 4.1 캐싱 전략

```python
from functools import lru_cache
import hashlib

class CachedRecommender:
    def __init__(self, recommender, cache_size: int = 100):
        self.recommender = recommender
        self._cache = {}

    def _get_cache_key(self, question: str, risk: str) -> str:
        content = f"{question}_{risk}"
        return hashlib.md5(content.encode()).hexdigest()

    def recommend(self, question: str, risk: str = None) -> dict:
        key = self._get_cache_key(question, risk or "none")
        if key not in self._cache:
            self._cache[key] = self.recommender.recommend(question, risk_preference=risk)
        return self._cache[key]
```

### 4.2 성능 벤치마크

```python
import time

def benchmark_search(queries: list, retriever, n_runs: int = 3):
    times = []
    for query in queries:
        for _ in range(n_runs):
            start = time.time()
            results = retriever.invoke(query)
            elapsed = time.time() - start
            times.append(elapsed)

    return {
        "avg_latency": sum(times) / len(times),
        "p95_latency": sorted(times)[int(len(times) * 0.95)],
        "total_queries": len(queries) * n_runs,
    }
```
